0350                   PUSH	 B
0360                   LDA       077AH                ; GET LAST TIME STAMP VALUE
a370                   MOV       B,A                  ; PUT  IT  IN B FOR NOW
0380                   MVI       A,020H               ; READ LOWER SECONDS RTC VALUE
0390                   OUT       QT~RTC~ADDR~PORT
0400                   NOP                            ; SMALL DELAY FOR SLOW RTC IC
0410                   IN        QI~RTC~DATA~PORT     ; READ THE COUNTER
0420 *  ANI  OFH  ; MASK OFF AND KEEP LOWER BITS
0430                   CMP       B                   ; STILL THE SAME VALUE?
0440                   JZ        DISPLAY~RTC~EXIT
0450                   STA       077AH               ; SAVE FOR UPDATE COMPARING
0460                   POP       B

0500 DISPLAY"TIME~NOW  EQU       DISPLAY~TIME
