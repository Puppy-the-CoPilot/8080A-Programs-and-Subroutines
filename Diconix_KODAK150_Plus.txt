0650 CENTRONICS~DATA  	EQU     IOTECH~U16~PORTA

0670 CENTRONICS~STATUS 	EQU     IOTECH~U16~PORTC ; LOWER NIBBLE
0680 CENTRONICS~BUSY  	EQU     00000001B        ; PIN 11 ON RIBBON CABLE
0690 CENTRONICS~ERROR 	EQU     10000010B        ; PIN 15 ON RIBBON CABLE
0700 CENTRONICS~ACK   	EQU     00001000B        ; PIN  10 ON RIBBON CABLE
0710 CENTRONICS~INIT  	EQU     00000110B        ; PIN 16 ON RIBBON CABLE
0720 CENTRONICS~STROBE 	EQU     10000000B        ;  PIN 1 ON RIBBON CABLE
0730 CENTRONICS~PAPEREMPTY   EQU 00000100B       ;  PIN 12 ON RIBBON CABLE
0740 *

0760 * EXERCISE PRINTER PORT CARD
0770 *
0780 *
0790                          ORG      OECOOH         ; COULD BE ANYWHERE
0800 *
0810                          LXI      SP,OE7FFH      ; OR USE OPERATING SYSTEM STACK
0820                          CALL     LDINIT
0830                          LXY      H,PRINTER~TEST~MESS
0840                          CALL     PRINTLINEOUT   ; SEND OUT THAT PRINTER STRING
0850                          HLT
0860 *                                                                                                                   .
0870 * SUBROUTINES
0880 *
0890 * LIST OUTPUT CHECKS FOR THE PRINTER ATTENTION
0900 * IF PRINTER ATTENDION IS NEEDED, THEN SET CARRY (C) ON RETURN
0910 *
0920 LDOUT                    CALL     LDATN            ; PRINTER ATTENTION  = NO PAPER
0930                          RC                        ; YES RETURN CARRY SET
0940 LDOUT1                   IN       CENTRONICS~STATUS  ; PRINTER READY?
0950                          ANI      CENTRONICS~BUSY  ; PRINTER READY FOR NEXT BYTE?
0960                          JINZ     LPOUT            ; READY=BUSY=0
0970                          MOV      A,B
0980 LDOUT2                   OUT      CENTRONICS~DATA
0990                          CALL     CENTRONICS~DELAY ; DATA SETUP TIME
1000                          MVI      A,0
1010                          OUT      CENTRONICS~CTRL  ; STROBE=0                  .
1020                          CALL     CENTRONICS~DELAY
1030                          MVI      A,OFCH           ; STROBE=1
1040                          OUT      CENTRONICS~CTRL
1050                          LXI      B,4*ONE~MILLISECOND
1060                          JMP      PRINTERVDELAY~LOOP  ; DATA HOLD TIMB
1070 LDOUT3                   NOP
1080                          RET
1000 *
1100 * PRINTER ATTENTION CHECK
1110 * CARRY SET = ATTENTION
1120 *
1130 LDATN                    NOP                        ; NOP TO ACTIVATE THIS ROUTINE
1140                          IN       CENTRONICS~STATUS
1150 CENTRONICS~PROBLEM EQU CENTRONICS~PAPER~EMPTY
1160                          ANI      CENTRONICS~PROBLEM ; ATTENTION NO PRINTER PAPER
1170                          RZ                         ; OK
1180                          STC                        ; SET CARRY FOR ATTN
1190                          RET                        ; DONE
1200 *                                       
1210 * INITIALIZE THE LIST DEVICE INTERFACE 8255 PARALLEL PORTS
1220 *                                       
1230 LDINIT                   MVI      A,10000011B       ; ONLY PORTC(0:3) ARE INPUTS
1240 INIT~CENTRONICS~PORT EQU LDINIT
1250                          OUT      IOTECH~U16~CTRL   ; SET UP THE 8255 IO PORTIc
1260                          MVI      A,OFCH            ; SET STROBE=1
1270                          OUT      CENTRONICS~CTRL
1280                          OUT      CENTRONICS~DATA
1290                          LXI      H,PRINTER~SETUP   ; RESET AND CONFIGURE THE PRINTER
1300                          CALL     PRINTLINEOUT
1310 DOUBLE~PRTR~DELAY        LXI      B,OFFFFH          ; DELAY BEFORE SENDING PRINT DATA
1320                          JMP      PRINTER~DELAY~LOOP
1330 *
1340 CENTRONICS~DELAY         LXI      B,ONE~MILLISECOND/4  ; 250US STROBE DURATION
1350 PRINTER~DELAY            DCX      B
1360                          MOV      A,B               ; CONTINUE DELAY?
1370                          ORA      C
1380                          RZ
1390                          JMP      PRINTER~DELAY~LOOP                                                 .

1420 ******************************************************************
1430 PRINTLINEOUT             MOV      A,M                ; GET BYTE TO TRANSMIT FROM (HL)
1440                          ANI      O7FH               ; STRIP TC ONLY ASCII 7 BITS
1450                          MOV      B,A
1460                          CALL     LDOUT              ; SEND REG.B TO PRINTER
1470                          MOV      A,M                ; CHECK BIT7 OF BYTE SENT
1480                          ANT      080H               ; IS MSB SET?
1490                          RNZ                         ; YES, THEN STRING IS DONE
1500                          INX      H                  ; POINT AT NEXT BYTE
1510                          JMP      PRINTLINEOUT
